window.showToast = (msg, type = 'success') => {
  const c = document.getElementById('toast-container');
  const t = document.createElement('div');
  t.className = `toast ${type}`;
  const icons = { success: 'fa-check-circle', error: 'fa-exclamation-circle', warning: 'fa-exclamation-triangle' };

  setTimeout(() => { t.style.animation = 'slideUp 0.3s reverse forwards'; setTimeout(() => t.remove(), 300); }, 3500);

window.togglePass = (id, btn) => {
  const inp = document.getElementById(id);
  const isPass = inp.type === 'password';

function redirectByRole(role) {
  showLoader();
  setTimeout(() => {
    window.location.href = role === 'admin' ? 'admin-panel.html' : 'user-panel.html';
  }, 800);
}

async function checkUserAndRedirect(user) {
  try {
    const snap = await get(ref(db, `users/${user.uid}`));
    let role = 'user';
    if (snap.exists()) {
      role = snap.val().role || 'user';
    } else {
      await set(ref(db, `users/${user.uid}`), {
        email: user.email, balance: 0, role: 'user',
        joined: new Date().toISOString(), createdAt: serverTimestamp()
      });
    }
    if (ADMIN_EMAILS.includes(user.email)) {
      role = 'admin';
      await update(ref(db, `users/${user.uid}`), { role: 'admin' });
    }
    redirectByRole(role);
  } catch (e) {
    console.error(e);
    showToast("Data load error", "error");
    hideLoader();
  }
}

onAuthStateChanged(auth, (user) => {
  if (user) checkUserAndRedirect(user);
  else hideLoader();
});

document.getElementById('login-form').addEventListener('submit', async (e) => {
  e.preventDefault();
  const email = document.getElementById('login-email').value.trim();
  const pass = document.getElementById('login-password').value;
  const username = document.getElementById('login-username').value.trim();
  const phone = document.getElementById('login-phone').value.trim();
  if (!email || !pass) return showToast("Email और Password भरें", "error");
  showLoader();
  try {
    const cred = await signInWithEmailAndPassword(auth, email, pass);
    if (username || phone) await update(ref(db, `users/${cred.user.uid}`), { username, phone });
    showToast("Login Successful! Redirecting...", "success");
  } catch (err) {
    hideLoader();
    const msg = err.message.replace("Firebase: ", "").replace(/\(auth.*\)/, "").trim();
    showToast(msg || "Login failed", "error");
  }
});

document.getElementById('signup-form').addEventListener('submit', async (e) => {
  e.preventDefault();
  const username = document.getElementById('signup-username').value.trim();
  const email = document.getElementById('signup-email').value.trim();
  const pass = document.getElementById('signup-password').value;
  if (!email || !pass || pass.length < 6) return showToast("Password कम से कम 6 अक्षर", "error");
  showLoader();
  try {
    const cred = await createUserWithEmailAndPassword(auth, email, pass);
    await set(ref(db, `users/${cred.user.uid}`), {
      email, username, balance: 0, role: 'user',
      joined: new Date().toISOString(), createdAt: serverTimestamp()
    });
    showToast("Account Created! Redirecting...", "success");
  } catch (err) {
    hideLoader();
    const msg = err.message.replace("Firebase: ", "").replace(/\(auth.*\)/, "").trim();
    showToast(msg || "Signup failed", "error");
  }
});

async function googleAuth() {
  showLoader();
  try {
    const provider = new GoogleAuthProvider();
    const result = await signInWithPopup(auth, provider);
    const user = result.user;
    const snap = await get(ref(db, `users/${user.uid}`));
    if (!snap.exists()) {
      await set(ref(db, `users/${user.uid}`), {
        email: user.email, username: user.displayName || '',
        balance: 0, role: 'user',
        joined: new Date().toISOString(), createdAt: serverTimestamp()
      });
    }
    showToast("Google Login Successful!", "success");
  } catch (err) {
    hideLoader();
